cmake_minimum_required(VERSION 3.24)

project(PhreakGuitar VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")

include(FetchContent)

# --- JUCE 8 (pinned release) -------------------------------------------------
FetchContent_Declare(
    JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG        8.0.15
    GIT_SHALLOW    TRUE
)
FetchContent_MakeAvailable(JUCE)

# --- Catch2 (tests) ----------------------------------------------------------
FetchContent_Declare(
    Catch2
    GIT_REPOSITORY https://github.com/catchorg/Catch2.git
    GIT_TAG        v3.7.1
    GIT_SHALLOW    TRUE
)
FetchContent_MakeAvailable(Catch2)

# --- Bundled assets (built-in IRs, generated by tools/make_ir.py) ------------
juce_add_binary_data(PhreakGuitarAssets SOURCES
    resources/irs/ir_tweed410.wav
    resources/irs/ir_brit412.wav
    resources/irs/ir_black112.wav
)

# --- Engine ------------------------------------------------------------------
# JUCE-free DSP core, shared by the plugin, the render harness, the tests and
# the browser (WASM) build. Nothing here may depend on JUCE.
add_library(phreakguitar_engine STATIC
    engine/dsp/Oversampler.cpp
    engine/amp/DriveStage.cpp
    engine/amp/ToneStack.cpp
    engine/amp/AmpChain.cpp
    engine/amp/PowerAmp.cpp
    engine/amp/PowerSupply.cpp
    engine/amp/CircuitAmp.cpp
    engine/fx/Compressor.cpp
    engine/fx/Delay.cpp
    engine/fx/Eq.cpp
    engine/fx/Reverb.cpp
    engine/fx/PhotoVibe.cpp
    engine/fx/Looper.cpp
    engine/circuit/Netlist.cpp
    engine/circuit/DKSolver.cpp
    engine/circuit/Pedal.cpp
)
target_include_directories(phreakguitar_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/engine)
target_compile_options(phreakguitar_engine PRIVATE -Wall -Wextra -Wshadow -Wconversion -Wno-sign-conversion)
# No fused multiply-add: WebAssembly has none, so native renders and tests
# round exactly like the browser engine does.
target_compile_options(phreakguitar_engine PRIVATE -ffp-contract=off)

# --- Native host -------------------------------------------------------------
# JUCE side of the native build: cab convolution + IR loading around the
# engine. INTERFACE target: sources compile into each consumer (plugin,
# render tool), avoiding duplicate JUCE module symbols that arise when JUCE
# modules are compiled into a static lib AND into the targets that link it.
add_library(phreakguitar_host INTERFACE)
target_sources(phreakguitar_host INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/src/host/CabConvolver.cpp
)
target_include_directories(phreakguitar_host INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(phreakguitar_host INTERFACE
    phreakguitar_engine
    juce::juce_dsp
    juce::juce_audio_formats
    PhreakGuitarAssets
)
target_compile_definitions(phreakguitar_host INTERFACE
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
)

# --- Plugin ------------------------------------------------------------------
juce_add_plugin(PhreakGuitar
    COMPANY_NAME            "PhreakGuitar"
    BUNDLE_ID               "com.phreakguitar.phreakguitar"
    PLUGIN_MANUFACTURER_CODE Phgt
    PLUGIN_CODE             Phrk
    FORMATS                 Standalone AU VST3
    PRODUCT_NAME            "PhreakGuitar"
    IS_SYNTH                FALSE
    NEEDS_MIDI_INPUT        FALSE
    NEEDS_MIDI_OUTPUT       FALSE
    MICROPHONE_PERMISSION_ENABLED TRUE
    MICROPHONE_PERMISSION_TEXT "PhreakGuitar needs audio input to process your guitar."
    COPY_PLUGIN_AFTER_BUILD TRUE
)

target_sources(PhreakGuitar PRIVATE
    src/plugin/PluginProcessor.cpp
    src/plugin/PluginEditor.cpp
)
target_link_libraries(PhreakGuitar
    PRIVATE
        phreakguitar_host
        juce::juce_audio_utils
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)
target_compile_definitions(PhreakGuitar
    PUBLIC
        JUCE_VST3_CAN_REPLACE_VST2=0
        JUCE_DISPLAY_SPLASH_SCREEN=0
)

# --- Offline render harness --------------------------------------------------
juce_add_console_app(render PRODUCT_NAME "render")
target_sources(render PRIVATE tools/render_main.cpp)
target_link_libraries(render PRIVATE
    phreakguitar_host
    juce::juce_recommended_config_flags
)

# --- Tests -------------------------------------------------------------------
enable_testing()
# Engine tests are plain C++: no JUCE, so they also prove the engine has
# no hidden JUCE dependency.
add_executable(dsp_tests tests/dsp_tests.cpp tests/circuit_tests.cpp tests/looper_tests.cpp)
target_link_libraries(dsp_tests PRIVATE
    phreakguitar_engine
    Catch2::Catch2WithMain
)
target_compile_definitions(dsp_tests PRIVATE PHREAK_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME dsp_tests COMMAND dsp_tests)
